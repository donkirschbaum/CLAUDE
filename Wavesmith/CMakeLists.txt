cmake_minimum_required(VERSION 3.22)

project(Wavesmith VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version" FORCE)
endif()

option(WAVESMITH_BUILD_TESTS "Build the headless preset test runner" OFF)

# --- JUCE ---------------------------------------------------------------
# Use a local checkout if JUCE_DIR is given, otherwise download it.
if(DEFINED JUCE_DIR AND EXISTS "${JUCE_DIR}/CMakeLists.txt")
    add_subdirectory(${JUCE_DIR} ${CMAKE_BINARY_DIR}/JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.4
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

# --- Presets (generated by tools/generate_presets.py) -------------------
set(PRESET_JSON ${CMAKE_CURRENT_SOURCE_DIR}/Resources/factory_presets.json)
if(NOT EXISTS ${PRESET_JSON})
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_presets.py ${PRESET_JSON}
        RESULT_VARIABLE PRESET_RESULT)
    if(NOT PRESET_RESULT EQUAL 0)
        message(FATAL_ERROR "Preset generation failed")
    endif()
endif()

juce_add_binary_data(WavesmithData SOURCES ${PRESET_JSON})

set(WAVESMITH_SOURCES
    Source/Wavetables.cpp
    Source/Voice.cpp
    Source/SynthEngine.cpp
    Source/Effects.cpp
    Source/PresetManager.cpp
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp)

set(WAVESMITH_MODULES
    juce::juce_audio_utils
    juce::juce_audio_processors
    juce::juce_dsp)

# --- Plugin ---------------------------------------------------------------
juce_add_plugin(Wavesmith
    COMPANY_NAME "Wavesmith Audio"
    BUNDLE_ID com.wavesmithaudio.wavesmith
    PLUGIN_MANUFACTURER_CODE Wvsm
    PLUGIN_CODE Wsm1
    FORMATS AU VST3 Standalone
    PRODUCT_NAME "Wavesmith"
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Synth
    AU_MAIN_TYPE kAudioUnitType_MusicDevice)

target_sources(Wavesmith PRIVATE ${WAVESMITH_SOURCES})

target_compile_definitions(Wavesmith PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0)

target_link_libraries(Wavesmith
    PRIVATE WavesmithData ${WAVESMITH_MODULES}
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)

# --- Headless test runner ------------------------------------------------------
if(WAVESMITH_BUILD_TESTS)
    juce_add_console_app(WavesmithTest PRODUCT_NAME "WavesmithTest")
    target_sources(WavesmithTest PRIVATE tests/PresetTest.cpp ${WAVESMITH_SOURCES})
    target_compile_definitions(WavesmithTest PRIVATE
        JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 WAVESMITH_TEST=1)
    target_link_libraries(WavesmithTest PRIVATE WavesmithData ${WAVESMITH_MODULES} juce::juce_recommended_config_flags)
endif()
